import React, { useRef, useEffect } from 'react';
import { useGLTF } from '@react-three/drei';
import * as THREE from 'three';

// Drop-in production GLTF/GLB loader with automatic PBR traversal
export function ProductModel({ color, finish }: { color: string; finish: { metalness: number; roughness: number } }) {
  const { scene } = useGLTF('/models/proto-x1.glb', true, true, (loader) => {
    loader.manager.onError = () => console.warn('GLTF fallback to procedural geometry');
  });

  useEffect(() => {
    scene?.traverse((child: any) => {
      if (child.isMesh && child.material) {
        child.material.color.set(color);
        child.material.metalness = finish.metalness;
        child.material.roughness = finish.roughness;
        child.castShadow = true;
        child.receiveShadow = true;
      }
    });
  }, [color, finish, scene]);

  return <primitive object={scene} dispose={null} />;
}
